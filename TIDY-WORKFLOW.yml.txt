name: tidy
# ONE-CLICK CLEANUP, no terminal required. Actions tab, choose "tidy", press "Run workflow".
#
# A zip can only add or overwrite, so files removed from the project cannot leave the repo by upload.
# This deletes them properly and commits the result. Everything listed is already retired: inert stubs
# that exist only because they could not be deleted.
on:
  workflow_dispatch: {}

permissions:
  contents: write

jobs:
  tidy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Remove retired files and directories
        run: |
          git rm -rq --ignore-unmatch \
            app/api/analyst \
            components/AskAnalyst.jsx \
            tests/analyst.test.mjs \
            supabase/migration-019.sql \
            lib/harness.mjs \
            tests/harness.test.mjs \
            jobs/projection_run.mjs \
            app/legacy \
            mockups \
            workflows-for-github
      - name: Check nothing still references them
        run: |
          if grep -rn "AskAnalyst\|lib/harness\|app/legacy\|projection_run" \
              --include='*.jsx' --include='*.js' --include='*.mjs' \
              app components lib jobs tests 2>/dev/null | grep -v RETIRED; then
            echo "Still referenced. Refusing to commit a broken tree."
            exit 1
          fi
      - name: Verify the tree still passes
        run: |
          npm install
          npm test
      - name: Commit
        run: |
          git config user.name "github-actions"
          git config user.email "actions@github.com"
          if git diff --cached --quiet; then
            echo "Nothing to remove. Already tidy."
          else
            git commit -qm "tidy: remove retired files"
            git push
          fi
